cmake_minimum_required(VERSION 3.20)
project(executor_tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
enable_testing()
add_executable(executor_tests executor_tests.cpp ../../src/main_thread_executor.cpp)
target_include_directories(executor_tests PRIVATE ../../include)
target_compile_definitions(executor_tests PRIVATE NOMINMAX UNICODE _UNICODE)
target_link_libraries(executor_tests PRIVATE user32 comctl32)
if(MSVC)
    target_compile_options(executor_tests PRIVATE /EHsc)
endif()
add_test(NAME executor_lifecycle COMMAND executor_tests)
set_tests_properties(executor_lifecycle PROPERTIES TIMEOUT 60)
